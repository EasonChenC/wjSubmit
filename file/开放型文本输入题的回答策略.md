 那如果有文本输入题呢 这种题型如何根据问题生成随机答案

● 非常好的问题！文本输入题是最具挑战性的题型，因为需要生成符合语义的文本内容，而不是简单的数字选择。

  让我详细讲解文本题的答案生成策略。

---
  文本题的三种类型

  1. 结构化文本（可枚举）

  - 例如：城市、姓名、公司名、商品类别
  - 特点：有固定的候选池
  - 生成方式：从预定义列表随机选择

  2. 半结构化文本（模板化）

  - 例如：联系方式、邮箱、地址
  - 特点：有固定格式模式
  - 生成方式：使用Faker库或模板

  3. 开放文本（自由输入）

  - 例如：意见建议、问题描述、评论
  - 特点：无固定格式，需要语义理解
  - 生成方式：模板库 + LLM生成

  策略1：结构化文本 - 候选池

  识别方式

  通过题目文本关键词识别：

  class TextQuestionClassifier:
      """文本题分类器"""

      def classify_text_question(self, question: Question) -> str:
          """根据题目文本判断文本类型"""
          label = question.label.lower()
    
          # 城市类
          if any(kw in label for kw in ['城市', '地区', 'city', '所在地', '居住']):
              return 'city'
    
          # 姓名类
          elif any(kw in label for kw in ['姓名', 'name', '称呼', '名字']):
              return 'name'
    
          # 公司/单位类
          elif any(kw in label for kw in ['公司', '单位', 'company', '企业', '组织']):
              return 'company'
    
          # 职业类
          elif any(kw in label for kw in ['职业', '工作', 'job', 'occupation', '岗位']):
              return 'occupation'
    
          # 商品/产品类
          elif any(kw in label for kw in ['商品', '产品', 'product', '类别', '品类']):
              return 'product_category'
    
          # 联系方式
          elif any(kw in label for kw in ['手机', '电话', 'phone', 'mobile', '联系方式']):
              return 'phone'
    
          elif any(kw in label for kw in ['邮箱', 'email', '邮件']):
              return 'email'
    
          # 默认：开放文本
          else:
              return 'open_text'

      @classmethod
      def get_pool(cls, text_type: str) -> List[str]:
          """获取对应类型的候选池"""
          mapping = {

              'product_category': cls.PRODUCT_CATEGORIES,
              'occupation': cls.OCCUPATIONS,
              'company': cls.COMPANIES,
          }
          return mapping.get(text_type, [])

  from faker import Faker

  class FakerTextGenerator:
      """使用Faker生成半结构化文本"""

      def __init__(self):
          self.fake = Faker('zh_CN')  # 中文
    
      def generate(self, text_type: str) -> str:
          """根据类型生成文本"""
    
          if text_type == 'name':
              return self.fake.name()  # 张伟
    
          elif text_type == 'phone':
              return self.fake.phone_number()  # 13812345678
    
          elif text_type == 'email':
              return self.fake.email()  # zhangwei@example.com
    
          elif text_type == 'company':
              return self.fake.company()  # 某某科技有限公司
    
          elif text_type == 'address':
              return self.fake.address()  # 北京市朝阳区某街道123号
    
          elif text_type == 'city':
              return self.fake.city()  # 随机城市
    
          elif text_type == 'url':
              return self.fake.url()  # https://example.com
    
          elif text_type == 'date':
              return self.fake.date()  # 2024-01-15
    
          else:
              # 默认：短句子
              return self.fake.sentence()

  根据前面答案生成逻辑一致的文本

  class ContextAwareTextGenerator:
      """上下文感知的文本生成器"""

      def generate_feedback(self, previous_answers: Dict) -> str:
          """根据之前的评分生成一致的反馈"""
    
          # 检查满意度评分（假设q10是满意度）
          satisfaction = previous_answers.get('q10', 3)
    
          if satisfaction >= 4:
              # 高分 → 正面评价
              template = random.choice(OpenTextTemplateLibrary.FEEDBACK_POSITIVE)
          elif satisfaction <= 2:
              # 低分 → 负面评价
              template = random.choice(OpenTextTemplateLibrary.FEEDBACK_NEGATIVE)
          else:
              # 中等 → 中性评价
              template = random.choice(OpenTextTemplateLibrary.FEEDBACK_NEUTRAL)
    
          return template
    
      def generate_suggestion(self, question_label: str) -> str:
          """根据题目类型生成建议"""
          label = question_label.lower()
    
          if '改进' in label or '建议' in label:
              return random.choice(OpenTextTemplateLibrary.SUGGESTIONS)
    
          elif '原因' in label or '理由' in label:
              return random.choice(OpenTextTemplateLibrary.REASONS)
    
          else:
              # 默认：正面反馈
              return random.choice(OpenTextTemplateLibrary.FEEDBACK_POSITIVE)

  策略4：LLM生成（高质量开放文本）

  当需要更自然的文本时

  class LLMTextGenerator:
      """使用LLM生成开放文本"""

      def __init__(self, llm_client):
          self.llm = llm_client
    
      async def generate_text(self, question: Question, context: Dict) -> str:
          """根据问题和上下文生成文本"""
    
          prompt = f"""
  请生成一段自然的中文回答。

  题目: {question.label}
  上下文:
  - 之前的满意度评分: {context.get('q10', '未知')}
  - 购物频率: {context.get('q8', '未知')}

  要求:
  1. 语言自然流畅
  2. 与上下文逻辑一致
  3. 20-50字
  4. 避免重复和模板化

  示例: "整体体验不错，物流速度快，商品质量也符合预期，会推荐给朋友。"

  请生成回答:
  """

          response = await self.llm.generate(prompt)
          return response.strip()

  完整的动态文本生成器

  class DynamicTextAnswerGenerator:
      """动态文本答案生成器"""

      def __init__(self, use_llm=False):
          self.classifier = TextQuestionClassifier()
          self.faker = FakerTextGenerator()
          self.context_aware = ContextAwareTextGenerator()
          self.use_llm = use_llm
          if use_llm:
              self.llm_generator = LLMTextGenerator(llm_client)
    
      def generate_text_answer(
          self,
          question: Question,
          previous_answers: Dict = None
      ) -> str:
          """生成文本答案的统一入口"""
    
          # 1. 分类文本题类型
          text_type = self.classifier.classify_text_question(question)
    
          # 2. 根据类型选择生成策略
          if text_type in ['city', 'product_category', 'occupation']:
              # 结构化文本：从候选池选择
              pool = TextCandidatePool.get_pool(text_type)
              if pool:
                  return random.choice(pool)
    
          if text_type in ['name', 'phone', 'email', 'company', 'address']:
              # 半结构化文本：使用Faker
              return self.faker.generate(text_type)
    
          if text_type == 'open_text':
              # 开放文本
              if self.use_llm:
                  # 策略A: LLM生成（高质量）
                  return await self.llm_generator.generate_text(
                      question,
                      previous_answers or {}
                  )
              else:
                  # 策略B: 模板库（快速）
                  return self.context_aware.generate_feedback(
                      previous_answers or {}
                  )
    
          # 默认：Faker生成短句
          return self.faker.fake.sentence()

| 策略    | 生成质量 | 速度 | 成本   | 适用场景           |
| ------- | -------- | ---- | ------ | ------------------ |

---
  总结

  文本题答案生成的四级策略：

  1. L1: 候选池 → 结构化文本（城市、类别）
  2. L2: Faker → 半结构化文本（姓名、电话）
  3. L3: 模板库 → 开放文本快速生成（评论、建议）
  4. L4: LLM → 开放文本高质量生成（复杂语义）

  智能降级：
  - 默认使用L1-L3（零成本）
  - 需要高质量时才使用L4（有成本）
  - 根据题目重要性和问卷预算动态选择

  这样，无论什么类型的文本题，系统都能智能生成合理的答案！
